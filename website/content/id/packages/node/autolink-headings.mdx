```ts
import { autolinkHeadings } from "@unifast/node";
```

## Signature

```ts
function autolinkHeadings(options?: AutolinkHeadingsOptions): UnifastPlugin
```

## Parameter

### options?

Konfigurasi untuk perilaku link heading

| Properti | Tipe | Default | Deskripsi |
|----------|------|---------|-------------|
| `behavior` | `"prepend" \| "append" \| "wrap"` | `"prepend"` | Posisi penempatan anchor link relatif terhadap konten heading |

## Penggunaan

```ts
import { compile, autolinkHeadings } from "@unifast/node";

const md = `# Introduction

## Getting Started

### Installation`;

const result = compile(md, {
  plugins: [
    autolinkHeadings({
      behavior: "prepend", // "prepend" | "append" | "wrap"
    }),
  ],
});
```

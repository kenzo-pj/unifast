```ts
import { externalLinks } from "@unifast/node";
```

## Signature

```ts
function externalLinks(options?: ExternalLinksPluginOptions): UnifastPlugin
```

## Parameter

### options?

Konfigurasi untuk atribut link eksternal

| Properti | Tipe | Default | Deskripsi |
|----------|------|---------|-------------|
| `rel` | `string` | `"nofollow noopener noreferrer"` | Nilai untuk atribut rel pada link eksternal |
| `target` | `string` | — | Nilai untuk atribut target (misalnya, "\_blank") |

## Penggunaan

```ts
import { compile, externalLinks } from "@unifast/node";

const result = compile(md, {
  plugins: [
    externalLinks({
      rel: "nofollow noopener noreferrer",
      target: "_blank",
    }),
  ],
});
```

```ts
import { emoji } from "@unifast/node";
```

## 函式簽名

```ts
function emoji(): UnifastPlugin
```

## 參數

無。

## 用法

```ts
import { compile, emoji } from "@unifast/node";

const md = `This is great :thumbsup: and I'm happy :smile:!`;

const result = compile(md, {
  plugins: [emoji()],
});
// :thumbsup: → 👍, :smile: → 😄
```

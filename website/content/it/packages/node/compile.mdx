---
title: "compile()"
description: "La funzione principale di compilazione. Trasforma Markdown o MDX in HTML o altri formati di output utilizzando il compilatore Rust nativo."
---

```ts
import { compile } from "@unifast/node";
```

## Firma

```ts
function compile(input: string, options?: CompileOptions): CompileResult
```

## Parametri

### input

| Proprietà | Tipo | Predefinito | Descrizione |
|-----------|------|-------------|-------------|
| `input` | `string` | — | Stringa sorgente Markdown o MDX |

### options?

| Proprietà | Tipo | Predefinito | Descrizione |
|-----------|------|-------------|-------------|
| `options?` | `CompileOptions` | — | Configurazione della compilazione, inclusi i plugin |

## Valore restituito

| Proprietà | Tipo | Descrizione |
|-----------|------|-------------|
| `output` | `string \| object` | Output compilato. Stringa HTML per impostazione predefinita; stringa JSON per HAST/MDAST. |
| `frontmatter` | `Record<string, unknown>` | Metadati del frontmatter analizzati (oggetto vuoto `{}` se assenti) |
| `diagnostics` | `Diagnostic[]` | Array di `{ level, message, start?, end? }` |
| `stats` | `{ parseMs, transformMs, emitMs }` | Dettaglio dei tempi (ms) |
| `toc` | `TocEntry[]` | Tabella dei contenuti estratta (array vuoto `[]` se il TOC è disabilitato) |

## Utilizzo

```ts
import { compile, gfm, frontmatter, toc, sanitize } from "@unifast/node";

const md = `---
title: My Document
---

# Hello, **world**!

| Column A | Column B |
|----------|----------|
| Cell 1   | Cell 2   |
`;

const result = compile(md, {
  inputKind: "md",       // "md" | "mdx"
  outputKind: "html",    // "html" | "hast" | "mdast" | "mdxJs"
  plugins: [gfm(), frontmatter(), toc({ maxDepth: 3 }), sanitize()],
});

console.log(result.output);
// <h1 id="hello-world">Hello, <strong>world</strong>!</h1>
// <table>...</table>

console.log(result.frontmatter);
// { title: "My Document" }

console.log(result.toc);
// [{ depth: 1, text: "Hello, world!", slug: "hello-world" }]

console.log(result.stats);
// { parseMs: 0.12, transformMs: 0.08, emitMs: 0.05 }
```

## Esempi

### Markdown di base in HTML

```ts
import { compile } from "@unifast/node";

const result = compile("# Hello, **world**!");

console.log(result.output);
// <h1 id="hello-world">Hello, <strong>world</strong>!</h1>
```

### Input MDX

```ts
import { compile } from "@unifast/node";

const mdx = `
import { Alert } from "./components";

# Hello

<Alert type="info">This is MDX!</Alert>
`;

const result = compile(mdx, { inputKind: "mdx" });
```

### Output HAST

```ts
import { compile } from "@unifast/node";

const result = compile("# Hello", { outputKind: "hast" });
const hast = JSON.parse(result.output as string);

console.log(hast);
// { type: "root", children: [{ type: "element", tagName: "h1", ... }] }
```

### Output MDAST

```ts
import { compile } from "@unifast/node";

const result = compile("# Hello", { outputKind: "mdast" });
const mdast = JSON.parse(result.output as string);

console.log(mdast);
// { type: "root", children: [{ type: "heading", depth: 1, ... }] }
```

### Output MDX JS

```ts
import { compile } from "@unifast/node";

const result = compile(mdxSource, {
  inputKind: "mdx",
  outputKind: "mdxJs",
});

console.log(result.output);
// JavaScript module string with MDX compiled to JSX
```

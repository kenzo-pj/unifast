## Signature

```ts
function rubyAnnotation(): UnifastPlugin
```

## Parameters

कोई नहीं।

## उपयोग

```ts
import { compile, rubyAnnotation } from "@unifast/node";

const md = `
{漢字|かんじ}の読み方を学びましょう。
`;

const result = compile(md, { plugins: [rubyAnnotation()] });
// {漢字|かんじ} <ruby>漢字<rp>(</rp><rt>かんじ</rt><rp>)</rp></ruby> बन जाता है
```
